% Apéndice APA: \apendice{A}{Título del apéndice}
\newcommand{\apendice}[2]{%
  \clearpage
  \section*{Apéndice #1}%
  \addcontentsline{toc}{section}{Apéndice #1}%
  \subsection*{#2}%
  \setcounter{table}{0}%
  \renewcommand{\thetable}{#1\arabic{table}}%
  \setcounter{figure}{0}%
  \renewcommand{\thefigure}{#1\arabic{figure}}%
}

% --- Portada APA ---
\newcommand{\PortadaAPA}{%
  \ifPortadaProfesional
    \fancyhead[L]{\MakeUppercase{\RunningHead}}%
  \fi
  \begin{titlepage}
  \thispagestyle{fancy}
  \centering
  \vspace*{2.5cm}
  {\bfseries\large \Titulo\par}
  \vspace{2cm}
  {\normalsize \AutorA\par}
  \ifdefempty{\AutorB}{}{{\normalsize \AutorB\par}}
  \vspace{1.5cm}
  {\normalsize \Filiacion\par}
  {\normalsize \Curso\par}
  {\normalsize \Docente\par}
  \vspace{1cm}
  {\normalsize \Fecha\par}
  \end{titlepage}
}
